cmake_minimum_required(VERSION 3.20)
project(PassScienceWorkbench VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(WorkbenchCLI src/main.cpp)
target_link_libraries(WorkbenchCLI PRIVATE Threads::Threads)
add_executable(Launch WIN32 src/launcher.cpp)
target_link_libraries(Launch PRIVATE Threads::Threads ws2_32 shell32 advapi32)
add_custom_command(TARGET Launch POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  ${CMAKE_SOURCE_DIR}/ui $<TARGET_FILE_DIR:Launch>/ui)
foreach(target WorkbenchCLI Launch)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /O2 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -O2 -static -static-libgcc -static-libstdc++)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endforeach()
